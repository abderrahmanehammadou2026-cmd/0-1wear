        function handleImageFileSelected(event) {
            const file = event.target.files[0];
            if (!file || !currentUploadProductId) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const img = new Image();
                img.onload = function() {
                    const canvas = document.createElement('canvas');
                    const maxDim = 750;
                    let w = img.width;
                    let h = img.height;
                    if (w > h && w > maxDim) {
                        h = Math.round((h * maxDim) / w);
                        w = maxDim;
                    } else if (h > maxDim) {
                        w = Math.round((w * maxDim) / h);
                        h = maxDim;
                    }
                    canvas.width = w;
                    canvas.height = h;
                    const ctx = canvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, w, h);
                    const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.84);

                    try {
                        customProductImages[currentUploadProductId] = compressedDataUrl;
                        localStorage.setItem('01wear_custom_images', JSON.stringify(customProductImages));
                        renderProducts();
                        if (!document.getElementById('photoStudioModal').classList.contains('hidden')) {
                            renderPhotoStudioCards();
                        }
                        showToast('تم حفظ وإدراج صورة المنتج بنجاح!');
                    } catch (err) {
                        showToast('حجم الصورة كبير جداً، جرب صورة أخرى.');
                    }
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
            event.target.value = '';
        }

        // Add Item to Cart
        function addToCart(productId, openCheckoutImmediately = false) {
            const product = products.find(p => p.id === productId);
            if (!product) return;

            const chosenSize = product.isBundle
                ? `الملابس: ${product.selectedClothingSize} | الحذاء: ${product.selectedShoeSize}`
                : product.selectedSize;

            const cartKey = `${product.id}_${chosenSize}`;
            const existing = cart.find(item => item.cartKey === cartKey);

            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({
                    cartKey,
                    id: product.id,
                    title: product.title,
                    price: product.price,
                    size: chosenSize,
                    qty: 1
                });
            }

            updateCartUI();

            if (openCheckoutImmediately) {
                toggleCart(true);
            } else {
                showToast(`تمت إضافة «${product.title}» إلى السلة`);
            }
        }

        // Change Quantity in Cart
        function changeQty(cartKey, delta) {
            const idx = cart.findIndex(i => i.cartKey === cartKey);
            if (idx === -1) return;

            cart[idx].qty += delta;
            if (cart[idx].qty <= 0) {
                cart.splice(idx, 1);
            }
            updateCartUI();
        }

            document.getElementById('cartShipping').textContent = shippingFee === 0 ? 'توصيل مجاني' : `${shippingFee} درهم`;

        }

                backdrop.classList.add('opacity-0');

            }
        }

            const grandTotal = subtotal + shippingFee;

            const itemsLines = cart.map((item, idx) =>
                `${idx + 1}. ${item.title}\n   • المقاس: ${item.size} | الكمية: ${item.qty} | المجموع: ${item.price * item.qty} درهم`
            ).join('\n');

            const shippingLine = shippingFee === 0 ? 'مجاني (0 درهم)' : `${shippingFee} درهم`;

            const message = `🛍️ *طلب جديد — متجر 0/1 WEAR*\n` +
                `----------------------------------\n` +
                `👤 *الاسم الكامل:* ${name}\n` +
                `📞 *رقم الهاتف:* ${phone}\n` +
                `🏙️ *المدينة:* ${city}\n` +
                `📍 *العنوان:* ${address}\n` +
                (notes ? `📝 *ملاحظة:* ${notes}\n` : '') +
                `----------------------------------\n` +
                `📦 *المنتجات المطلوبة:*\n${itemsLines}\n` +
                `----------------------------------\n` +
                `🚚 *التوصيل:* ${shippingLine}\n` +
                `💰 *المبلغ الإجمالي للدفع عند الاستلام:* ${grandTotal} درهم`;

            lastReceiptText = message;
            document.getElementById('receiptSummaryBox').textContent = message;

            const cleanWa = sellerWhatsapp.replace(/[^0-9]/g, '');
            const waUrl = `https://wa.me/${cleanWa}?text=${encodeURIComponent(message)}`;
            document.getElementById('whatsappDirectLink').href = waUrl;

            toggleCart(false);
            const modal = document.getElementById('receiptModal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');

            window.open(waUrl, '_blank');
        }

        function closeReceiptModal() {
            const modal = document.getElementById('receiptModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        function copyReceiptText() {
            const temp = document.createElement('textarea');
            temp.value = lastReceiptText;
            document.body.appendChild(temp);
            temp.select();
            document.execCommand('copy');
            document.body.removeChild(temp);
            showToast('تم نسخ تفاصيل الطلب بنجاح!');
        }

        // Seller Settings Modal
        function toggleSellerModal() {
            const modal = document.getElementById('sellerModal');
            modal.classList.toggle('hidden');
            modal.classList.toggle('flex');
        }

        function saveSellerSettings() {
            const waInput = document.getElementById('sellerWhatsappInput').value.trim();
            const shipInput = parseInt(document.getElementById('sellerShippingInput').value || '0', 10);
            const photoBtnChecked = document.getElementById('adminPhotoToggleCheckbox').checked;

            if (waInput) {
                sellerWhatsapp = waInput;
                localStorage.setItem('01wear_seller_wa', sellerWhatsapp);
            }
            shippingFee = isNaN(shipInput) || shipInput < 0 ? 0 : shipInput;
            localStorage.setItem('01wear_shipping_fee', shippingFee);

            showCardPhotoBtn = photoBtnChecked;
            localStorage.setItem('01wear_show_card_photo', showCardPhotoBtn);

            renderProducts();
            updateCartUI();
            toggleSellerModal();
            showToast('تم حفظ إعدادات المتجر بنجاح!');
        }

        // Toast Helper
        function showToast(msg) {
            const toast = document.getElementById('toastNotification');
            document.getElementById('toastMessage').textContent = msg;
            toast.classList.remove('translate-y-20', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 2600);
        }
